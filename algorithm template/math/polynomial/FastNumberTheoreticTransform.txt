mint qpow(mint a,int b)
{
    mint res=1;
    while(b)
    {
        if(b&1)res*=a;
        a*=a;
        b>>=1;
    }
    return res;
}
using Poly=vector<mint>;
void ntt(Poly& f,bool inv)
{
    int n=f.size();
    for(int i=1,j=0;i<n;++i)
    {
        int bit=n>>1;
        for(;j&bit;bit>>=1)j^=bit;
        j^=bit;
        if(i<j)swap(f[i],f[j]);
    }
    for(int len=1;len<n;len<<=1)
    {
        mint wn=qpow(inv?Gi:G,(mod-1)/(len<<1));
        for(int i=0;i<n;i+=len<<1)
        {
            mint w=1;
            for(int j=0;j<len;++j,w*=wn)
            {
                mint x=f[i+j];
                mint y=f[i+j+len]*w;
                f[i+j]=x+y;
                f[i+j+len]=x-y;
            }
        }
    }
    if(inv)
    {
        mint invn=qpow(n,mod-2);
        for(mint& v:f)v*=invn;
    }
}
