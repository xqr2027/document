auto manacher(const string& t)// 0 indexed
{
    string s="`~";
    for(auto c:t)s+=c,s+='~';
    int n=s.size()-1;
    vector<int>d(n+1);
    int l=1,r=0;
    for(int i=1;i<=n;++i)
    {
        if(i<=r)d[i]=min(d[l+r-i],r-i+1);
        while(i+d[i]<=n&&s[i+d[i]]==s[i-d[i]])++d[i];
        if(i+d[i]-1>r)l=i-(d[i]-1),r=i+(d[i]-1);
    }
    return d;
}
