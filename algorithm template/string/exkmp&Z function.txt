#include<bits/stdc++.h>
using namespace std;
#define int long long
auto exkmp(const string& s)// 1 indexed
{
    int n=s.size()-1;
    vector<int>z(n+1);
    z[1]=n;
    int l=1,r=0;
    for(int i=2;i<=n;++i)
    {
        if(i<=r)z[i]=min(z[i-l+1],r-i+1);
        while(i+z[i]<=n&&s[1+z[i]]==s[i+z[i]])++z[i];
        if(i+z[i]-1>r)l=i,r=i+z[i]-1;
    }
    return z;
}
