import matplotlib.pyplot as plt
plt.rcParams['font.sans-serif']=['SimHei','Noto Sans CJK SC','Microsoft YaHei']
plt.rcParams['axes.unicode_minus']=False
fig,ax=plt.subplots(figsize=(7,7))
for t in range(-2,7): ax.axvline(t,color='#eeeeee',zorder=0)
for t in range(0,7):  ax.axhline(t,color='#eeeeee',zorder=0)
ax.axhline(0,color='k',lw=.8); ax.axvline(0,color='k',lw=.8)
# 镜面 y=x+1（红虚线）
ax.plot([-2,5.5],[-1,6.5],'r--',lw=1.5,label='镜面 y=x+1')
# 翻转前前缀（虚线橙）：A(0,0)->(1,0)->(2,0)->(2,1)->(2,2)->P1(2,3)
ax.plot([0,1,2,2,2,2],[0,0,0,1,2,3],color='orange',ls='--',lw=2.5,marker='o',label='翻转前（虚线，只到P1）')
# 翻转后前缀（实线橙）：A'(-1,1)->(-1,2)->(-1,3)->(0,3)->(1,3)->P1(2,3)
ax.plot([-1,-1,-1,0,1,2],[1,2,3,3,3,3],color='orange',ls='-',lw=2.5,marker='o',label='翻转后（实线，只到P1）')
# 不变后缀（实线青）：P1->...->P2->...->B(6,6)，仍经过第二次相交P2
ax.plot([2,3,4,4,4,5,6,6],[3,3,3,4,5,5,5,6],color='teal',ls='-',lw=2.5,marker='o',label='不变后缀（青，含P2）')
# 镜像双箭头 A<->A'
ax.annotate('',xy=(-1,1),xytext=(0,0),arrowprops=dict(arrowstyle='<->',color='gray',ls='--'))
# 标注
for x,y,s in [(0,0,'A(0,0)'),(6,6,'B(6,6)'),(-1,1,"A'(-1,1)")]:
    ax.annotate(s,(x,y),textcoords='offset points',xytext=(6,6))
ax.plot(2,3,'ko',ms=8); ax.annotate('P1(2,3) 第一次相交',(2,3),textcoords='offset points',xytext=(8,-14))
ax.plot(4,5,'k*',ms=14); ax.annotate('P2(4,5) 第二次相交',(4,5),textcoords='offset points',xytext=(8,6))
ax.set_xlim(-2.2,6.6); ax.set_ylim(-0.6,6.6); ax.set_aspect('equal')
ax.set_xticks(range(-2,7)); ax.set_yticks(range(0,7))
ax.legend(loc='lower right'); ax.set_title('6×6 反射原理：两次相交，只翻折第一次之前')
plt.tight_layout(); plt.show()
